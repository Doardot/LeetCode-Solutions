so now we have this:

sortByBits(vector<int> &arr)

// first we create a vector to store the val and count
vector<pair<int, int>> vecNumAndCount;
// then calls convertDecToBin and stores the value and the count in the vector
for (int val : arr)
{
    bin = convertDecToBin(val);
    vecNumAndCount.emplace_back(val, bin);
}

// then we sort with the premise that if the binCount is equal for both numbers
// we use the val to decide
sort(vecNumAndCount.begin(), vecNumAndCount.end(), [](const pair<int, int> &a, const pair<int, int> &b) {
    if (a.second == b.second) {
        return a.first < b.first;
    }
    return a.second < b.second;
});

//then we recreate the vector with the vals now ordered
vector<int> result;
for (auto &p : vecNumAndCount) 
{
    result.push_back(p.first);
}
