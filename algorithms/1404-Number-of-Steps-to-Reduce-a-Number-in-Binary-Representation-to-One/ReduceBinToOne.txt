first solution

int numSteps(string s)
{
    if (s == "1")
    {
        return 0;
    }

    int count = 0;
    for (int i = s.length() - 1; i > 0; --i)
    {
        if (s[i] == '0')
        {
            cout << s << "\n";
            s.erase(i);
            ++count;
            cout << "0: " << s << "\n";
        }
        else if (s[i] == '1')
        {
            cout << s << "\n";
            s[i] = '0';
            s[i + 1] = '1';
            count += 2;
            cout << "1: " << s << "\n";
        }
    }
    return count;
}

int count = 0;
int carry = 0;
for (int i = s.length() - 1; i > 0; --i)
{
    // ascii table: 0 = 48 | 1 = 49
    int bit = s[i] - '0';
    int sum = bit + carry;
    if (sum == 0)
    {
        ++count;
        carry = 0;
    }
    else if (sum == 1)
    {

        count += 2;
        carry = 1;
    }
    else if (sum == 2)
    {
        ++count;
        carry = 1;
    }
}
return count + carry;

if sum is 0: the bit is 0
just "divide" (move to the next bit from right to left) 
count += 1.

if sum is 1: 
either the bit is 1 or there is a carry
either way, the number is odd
we count the addition AND the division
count += 2.

but we dont need to change the bit in the string!! we are just "simulating" the calculus
